4. COMPLETE FIRESTORE SECURITY RULES
================================================================================
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAuthenticated() {
      return request.auth != null;
    }

    function isAdmin() {
      return isAuthenticated() && (
        request.auth.token.email == 'beulaaudio@gmail.com' ||
        request.auth.token.admin == true ||
        (exists(/databases/$(database)/documents/admins/$(request.auth.uid)) &&
         get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.enabled == true)
      );
    }

    function isBookingOwner(resData) {
      return isAuthenticated() && (
        resData.userId == request.auth.uid ||
        resData.customer.phone == request.auth.token.phone_number ||
        (request.auth.token.email != null && resData.userEmail == request.auth.token.email)
      );
    }

    function isValidStatus(status) {
      return status in ['pending', 'confirmed', 'cancelled', 'completed', 'rejected'];
    }

    function isValidStageStatus(stage) {
      return stage in ['ordered', 'accepted', 'declined', 'shipped', 'on_live', 'completed'];
    }

    function isValidCategory(category) {
      return category in ['dj', 'instrument', 'custom'];
    }

    function isValidEventType(type) {
      return type in ['marriage', 'birthday', 'ear_piercing', 'corporate', 'others'];
    }

    match /settings/{settingId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /packages/{packageId} {
      allow get, list: if true;
      allow create, update, delete: if isAdmin();
    }

    match /packages/{packageId}/{subcollection}/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /inventory/{docId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /inventory_items/{itemId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /bookings/{bookingId} {
      allow get: if true;
      allow list: if isAdmin() || (isAuthenticated() && resource.data.userId == request.auth.uid);
      allow create: if 
        isAdmin() ||
        (
          isAuthenticated() &&
          request.resource.data.source == 'website' &&
          request.resource.data.userId == request.auth.uid &&
          request.resource.data.requestId is string &&
          request.resource.data.requestId.size() >= 6 &&
          request.resource.data.customer.name is string &&
          request.resource.data.customer.name.size() >= 2 &&
          request.resource.data.customer.phone is string &&
          request.resource.data.customer.phone.size() >= 10 &&
          request.resource.data.event.date is string &&
          request.resource.data.event.district is string &&
          request.resource.data.event.address is string &&
          isValidEventType(request.resource.data.event.type) &&
          isValidCategory(request.resource.data.setup.category) &&
          (request.resource.data.totalPriceBeforeTransport == null ||
           (request.resource.data.totalPriceBeforeTransport is number &&
            request.resource.data.totalPriceBeforeTransport >= 0))
        );

      allow update: if 
        isAdmin() ||
        (
          isBookingOwner(resource.data) &&
          resource.data.status == 'pending' &&
          request.resource.data.status == 'cancelled' &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'updatedAt', 'cancellationReason'])
        );

      allow delete: if isAdmin();
    }

    match /reviews/{reviewId} {
      // Public: individual gets allowed for all; listing only approved docs for guests
      allow get: if true;
      allow list: if isAdmin() || resource.data.approved == true;

      // Users can create with validated fields; userId must match own uid; approved=true enforced
      allow create: if isAuthenticated() &&
        request.resource.data.userName is string &&
        request.resource.data.userName.size() >= 2 &&
        request.resource.data.comment is string &&
        request.resource.data.comment.size() >= 5 &&
        request.resource.data.comment.size() <= 2000 &&
        request.resource.data.rating is number &&
        request.resource.data.rating >= 1 &&
        request.resource.data.rating <= 5 &&
        (!('userId' in request.resource.data) || request.resource.data.userId == request.auth.uid) &&
        request.resource.data.approved == true;

      // Admin: full update control; user: can edit own comment/rating but not approved status
      allow update: if isAdmin() ||
        (isAuthenticated() &&
         resource.data.userId == request.auth.uid &&
         request.resource.data.approved == resource.data.approved &&
         request.resource.data.userId == resource.data.userId);

      allow delete: if isAdmin() ||
        (isAuthenticated() && resource.data.userId == request.auth.uid);
    }

    match /bookingAudit/{auditId} {
      allow read: if isAdmin();
      allow create: if isAuthenticated() || isAdmin();
      allow update, delete: if isAdmin();
    }

    match /inventoryAudit/{auditId} {
      allow read, write: if isAdmin();
    }

    match /admins/{uid} {
      allow read: if isAuthenticated() && (request.auth.uid == uid || isAdmin());
      allow write: if isAdmin();
    }

    match /diagnostics/{docId} {
      allow read, write: if isAdmin();
    }

    match /{document=**} {
      allow read, write: if isAdmin();
    }
  }
}
